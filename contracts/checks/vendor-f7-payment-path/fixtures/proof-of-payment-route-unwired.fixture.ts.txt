import { NextRequest, NextResponse } from 'next/server';
import { getFirestore } from 'firebase-admin/firestore';
import { getStorage } from 'firebase-admin/storage';

import { getAdminSession } from '@/lib/admin-auth';
import { initAdmin } from '@/lib/firebase-admin';
import { VENDOR_SUBMISSIONS_COLLECTION } from '@/lib/vendor-submissions';

// KNOWN-UNWIRED FIXTURE -- four defeating mutations bundled: (1) no rate limiting at all --
// handleProofOfPaymentUpload() is never called, so no rate-limit gate exists; (2) upload
// metadata validation is hand-rolled inline instead of calling the real
// planProofOfPaymentUpload(); (3) existence LEAKS via a distinct 404 for a missing
// submission, defeating the non-enumerable posture; (4) it imports lib/admin-auth.ts,
// contradicting the "public, unauthenticated" design this route must have.
interface ProofOfPaymentRequestBody {
  fileName?: unknown;
  mimeType?: unknown;
  sizeBytes?: unknown;
  fileBase64?: unknown;
}

export async function POST(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> },
): Promise<NextResponse> {
  await getAdminSession(); // present but unused -- still counts as an import for the discriminator
  const { id } = await params;
  const body = (await request.json()) as ProofOfPaymentRequestBody;

  const storagePath = `vendor-proofs/${id}/${body.fileName}`;

  const db = getFirestore(initAdmin());
  const ref = db.collection(VENDOR_SUBMISSIONS_COLLECTION).doc(id);
  const snapshot = await ref.get();
  if (!snapshot.exists) {
    return NextResponse.json({ error: 'Vendor submission not found.' }, { status: 404 });
  }

  const bucket = getStorage(initAdmin()).bucket();
  const file = bucket.file(storagePath);
  await file.save(Buffer.from(typeof body.fileBase64 === 'string' ? body.fileBase64 : '', 'base64'));

  await ref.update({ proofOfPaymentPath: storagePath, proofOfPaymentUploadedAt: new Date() });

  return NextResponse.json({ success: true, storagePath });
}
