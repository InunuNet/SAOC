import { promises as dns } from 'node:dns';

import { NextRequest, NextResponse } from 'next/server';
import { Timestamp } from 'firebase-admin/firestore';

import {
  buildPayfastNotifyParamString,
  generateNotifySignature,
  getClientIp,
  PAYFAST_ITN_HOSTS,
  PAYFAST_SANDBOX_VALIDATE_URL,
} from '@/lib/payfast';
import { findReservedOrderByPaymentId, markOrderAndPositionPaidByPaymentId } from '@/lib/orders';
import { deliverConfirmationEmailAfterCommit, sendConfirmationEmail } from '@/lib/confirmation-email';

/**
 * PayFast ITN (Instant Transaction Notification) handler — SECURITY BOUNDARY, FAIL CLOSED.
 *
 * PayFast POSTs application/x-www-form-urlencoded on payment completion. Every check
 * below must pass before an order is ever flipped to 'paid'. Any single failure logs
 * which check failed and leaves the order untouched — we still return HTTP 200 so
 * PayFast stops retrying (see https://developers.payfast.co.za/docs#step_4_confirm_payment),
 * but a 200 response here never implies the payment was accepted.
 *
 * F10 (ticketing-foundation) — the single authorised reopening of this pinned file, folding
 * in four changes at once (see contracts/golden/ticketing-f10-itn-repin/README.md for the
 * full decision record and every defeating mutation):
 *   1. Inbound signature verification now uses the correct (posted-order, no-trim,
 *      no-blank-skip) algorithm via generateNotifySignature/buildPayfastNotifyParamString,
 *      not the outbound-only generateSignature/buildPayfastParamString this route used
 *      to reuse — see contracts/golden/payfast-itn-signature/ for why that was wrong.
 *   2. parseOrderedFields now BREAKS at the posted 'signature' field, matching PayFast's
 *      own documented reference implementation exactly, instead of CONTINUE-ing past it.
 *   3. A paid ITN now flips BOTH the order and its one child position to 'paid' inside a
 *      single Firestore transaction (lib/orders.ts, F2/F10) — the pre-F10 version of this
 *      route wrote only a flat 'tickets' document; that document is now the position, and
 *      the order (added in F2) is the payment source of truth.
 *   4. After a successful transaction commit, a try/catch-isolated call fires the
 *      confirmation-email hookup. An email failure is logged, never a reason to retry the
 *      write or return anything other than the normal 200 acknowledgement — the payment is
 *      already committed and must never be put at risk by a delivery-receipt failure.
 *
 * Amount-match tolerance below ZAR 0.01 (fixed cents rounding, not a config).
 */
const AMOUNT_MATCH_TOLERANCE = 0.01;

const COMPLETE_STATUS = 'COMPLETE';

/**
 * Resolve PayFast's published ITN source hosts (www.payfast.co.za, sandbox.payfast.co.za,
 * w1w.payfast.co.za, w2w.payfast.co.za — see PAYFAST_ITN_HOSTS) via DNS at request time.
 * IPs are never hardcoded; PayFast rotates them.
 */
async function resolvePayfastIps(): Promise<Set<string>> {
  const ips = new Set<string>();
  await Promise.all(
    PAYFAST_ITN_HOSTS.map(async (host) => {
      try {
        const results = await dns.lookup(host, { all: true });
        for (const result of results) ips.add(result.address);
      } catch (error) {
        console.error(`[tickets/itn] DNS lookup failed for host ${host}:`, error);
      }
    })
  );
  return ips;
}

/**
 * Parse the raw ITN body into (fields, signature), preserving POSTED order (URLSearchParams
 * iterates in the order it appears in the body) and excluding 'signature' from `fields`.
 *
 * F10: matches PayFast's own documented reference implementation exactly — iteration STOPS
 * (`break`) the instant the posted 'signature' key is reached, rather than skipping just
 * that one key and continuing (`continue`, the pre-F10 behaviour). PayFast always posts
 * `signature` last, so this is dormant in practice today, but a `continue`-based parse would
 * silently include any field that happened to arrive AFTER `signature` in the computed
 * digest — a digest PayFast's own `break`-based reference algorithm never included, so a
 * genuine, unmodified ITN would fail verification the moment PayFast's field order ever
 * changed. `break` removes that latent fragility and matches the spec exactly, not just
 * "whatever happens to work today". Exported (only change to its signature) so this exact
 * property is unit-testable without an HTTP round trip.
 */
export function parseOrderedFields(raw: string): { fields: Record<string, string>; signature: string | null } {
  const params = new URLSearchParams(raw);
  const fields: Record<string, string> = {};
  let signature: string | null = null;
  for (const [key, value] of params) {
    if (key === 'signature') {
      signature = value;
      break;
    }
    fields[key] = value;
  }
  return { fields, signature };
}

/** Always 200 — PayFast must stop retrying regardless of validation outcome. */
function acknowledge(): NextResponse {
  return NextResponse.json({ received: true }, { status: 200 });
}

export async function POST(request: NextRequest): Promise<NextResponse> {
  const rawBody = await request.text();
  const { fields, signature: receivedSignature } = parseOrderedFields(rawBody);
  const mPaymentId = fields['m_payment_id'] ?? null;

  // 1. Signature — recompute over the fields AS RECEIVED, in received order, excluding
  // the posted 'signature' field, using the INBOUND (verification) algorithm — PayFast
  // documents this as genuinely different from the outbound (checkout-signing) algorithm:
  // no blank-skip, no trim. See contracts/golden/payfast-itn-signature/inbound-algorithm.golden.md.
  const passphrase = process.env.PAYFAST_SANDBOX_PASSPHRASE;
  const expectedSignature = generateNotifySignature(fields, passphrase);
  if (!receivedSignature || expectedSignature !== receivedSignature) {
    console.error('[tickets/itn] Signature mismatch — rejecting ITN', { m_payment_id: mPaymentId });
    return acknowledge();
  }

  // 2. Source IP — request must originate from a resolved PayFast host
  // (www.payfast.co.za / sandbox.payfast.co.za / w1w.payfast.co.za / w2w.payfast.co.za).
  const clientIp = getClientIp(request);
  if (!clientIp) {
    console.error('[tickets/itn] Could not determine client IP — rejecting ITN', {
      m_payment_id: mPaymentId,
    });
    return acknowledge();
  }
  const validIps = await resolvePayfastIps();
  if (!validIps.has(clientIp)) {
    console.error('[tickets/itn] Source IP not in PayFast host set — rejecting ITN', {
      m_payment_id: mPaymentId,
      clientIp,
    });
    return acknowledge();
  }

  if (!mPaymentId) {
    console.error('[tickets/itn] Missing m_payment_id — rejecting ITN');
    return acknowledge();
  }

  // F10: order is the payment source of truth (F2) — look up the order by m_payment_id,
  // not the position. This read is NOT transactional (optimisation + amount-check only);
  // the authoritative re-read + write happens inside markOrderAndPositionPaidByPaymentId's
  // own transaction below. See that function (lib/orders.ts) for why this is safe under
  // concurrent/duplicate ITN delivery.
  let lookup: Awaited<ReturnType<typeof findReservedOrderByPaymentId>>;
  try {
    lookup = await findReservedOrderByPaymentId(mPaymentId);
  } catch (error) {
    console.error('[tickets/itn] Order lookup failed — rejecting ITN', {
      m_payment_id: mPaymentId,
      error,
    });
    return acknowledge();
  }

  if (lookup.status === 'not-found') {
    console.error('[tickets/itn] No order found for m_payment_id — rejecting ITN', {
      m_payment_id: mPaymentId,
    });
    return acknowledge();
  }

  // Idempotency fast path (optimisation only, NOT the correctness guarantee): if this read
  // already shows anything other than 'reserved', there is no transition for this ITN to
  // make, so skip the amount check and the external server-confirm round-trip. This read
  // is not transactional and can be stale under concurrent ITNs, so it must never be
  // relied on for the actual write decision — see the transactional status check inside
  // markOrderAndPositionPaidByPaymentId below.
  //
  // A non-'paid' short-circuit is logged, not silent: it means money may have moved for an
  // order this handler will not mark paid (a cancelled or malformed document), and the 200
  // below stops PayFast retrying. That is deliberate — fail closed on the write, loud in
  // the log, and let an operator reconcile — but it must be visible to reconcile.
  if (lookup.status === 'already-settled') {
    if (lookup.orderStatus !== 'paid') {
      console.error('[tickets/itn] Order is not reserved — ignoring ITN, no status change', {
        m_payment_id: mPaymentId,
        status: lookup.orderStatus,
      });
    }
    return acknowledge();
  }

  // 3. Amount match — the reserved order's server-derived amount must match what
  // PayFast says was actually charged.
  const amountGross = fields['amount_gross'];
  const orderAmount = lookup.amount;
  if (
    amountGross === undefined ||
    Number.isNaN(Number(amountGross)) ||
    Number.isNaN(orderAmount) ||
    Math.abs(Number(amountGross) - orderAmount) >= AMOUNT_MATCH_TOLERANCE
  ) {
    console.error('[tickets/itn] amount_gross does not match reserved order — rejecting ITN', {
      m_payment_id: mPaymentId,
      amountGross,
      orderAmount,
    });
    return acknowledge();
  }

  // 4. Server confirmation — POST the received data back to PayFast's
  // https://sandbox.payfast.co.za/eng/query/validate endpoint and require the response
  // body to equal exactly 'VALID'. Uses the SAME inbound param string as the signature
  // check (PayFast's docs specify one inbound param string, reused for both purposes).
  try {
    const confirmResponse = await fetch(PAYFAST_SANDBOX_VALIDATE_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: buildPayfastNotifyParamString(fields),
    });
    const confirmText = (await confirmResponse.text()).trim();
    if (confirmText !== 'VALID') {
      console.error('[tickets/itn] Server confirmation did not return VALID — rejecting ITN', {
        m_payment_id: mPaymentId,
        confirmStatus: confirmResponse.status,
      });
      return acknowledge();
    }
  } catch (error) {
    console.error('[tickets/itn] Server confirmation request failed — rejecting ITN', {
      m_payment_id: mPaymentId,
      error,
    });
    return acknowledge();
  }

  // All four checks passed. Only flip to 'paid' if PayFast also reports COMPLETE.
  if (fields['payment_status'] !== COMPLETE_STATUS) {
    console.error('[tickets/itn] payment_status is not COMPLETE — leaving order reserved', {
      m_payment_id: mPaymentId,
      payment_status: fields['payment_status'],
    });
    return acknowledge();
  }

  // 5. Atomic idempotent order+position two-write. PayFast retries ITN delivery until it
  // receives HTTP 200, and can genuinely deliver the same valid notification twice in close
  // succession. markOrderAndPositionPaidByPaymentId (lib/orders.ts) re-reads the order fresh
  // inside a single Firestore transaction, checks it is still 'reserved', and — only then —
  // writes 'paid' to BOTH the order and its one child position in that same transaction, so
  // whichever ITN commits first wins and a loser (duplicate delivery, or a genuine
  // concurrent ITN) sees the now-'paid' order and no-ops instead of writing again. See
  // lib/orders.ts and the golden README's "Idempotency" section for the full reasoning,
  // including why this is an UPDATE-based (not create-based) idempotency guard.
  const outcome = await markOrderAndPositionPaidByPaymentId({
    m_payment_id: mPaymentId,
    gatewayPaymentId: fields['pf_payment_id'] ?? null,
    now: Timestamp.now(),
  });

  if (!outcome.committed) {
    console.error('[tickets/itn] Transactional order+position update did not commit', {
      m_payment_id: mPaymentId,
      reason: outcome.reason,
    });
    return acknowledge();
  }

  // Post-commit confirmation email hookup (F10). F11 owns QR generation and the email's
  // actual content — sendConfirmationEmail (lib/confirmation-email.ts) is a minimal stub
  // until F11 lands. deliverConfirmationEmailAfterCommit isolates any failure from this
  // response: the payment is already committed above and must never be rolled back, retried,
  // or hidden behind a 5xx because a delivery receipt failed to send.
  await deliverConfirmationEmailAfterCommit(
    () =>
      sendConfirmationEmail({
        orderId: outcome.orderId,
        buyerEmail: outcome.buyerEmail,
        buyerName: outcome.buyerName,
        recoveryToken: outcome.recoveryToken,
        positions: [outcome.position],
      }),
    (error) => {
      console.error('[tickets/itn] Confirmation email failed — payment already committed, not rolled back', {
        m_payment_id: mPaymentId,
        orderId: outcome.orderId,
        error,
      });
    }
  );

  return acknowledge();
}
