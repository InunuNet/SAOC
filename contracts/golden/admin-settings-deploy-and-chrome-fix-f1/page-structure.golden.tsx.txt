// Not a literal file to copy-paste (the existing toggle UI, useState/useEffect fetch logic,
// and 'use client' directive from the current file are UNCHANGED) — this shows the required
// chrome wrapper and the new capability-derivation that page.tsx must gain, in the same shape
// app/admin/page.tsx and app/admin/vendors/page.tsx already use.
//
// THE CORE PROBLEM THIS GOLDEN FIXES: app/admin/settings/page.tsx is 'use client' (it needs
// useState/useEffect for the toggle). UtilityBar/Header/Footer/AdminNav's canManagePaymentSettings
// derivation are SERVER-side concerns (sanityFetch, hasCapability, getAdminSession — none of
// which may run in a client component). app/admin/page.tsx and app/admin/vendors/page.tsx are
// both `async function` Server Components that fetch and render chrome directly because a
// shared app/admin/layout.tsx would also wrap /admin/login and /admin/door (see the existing
// comment in app/admin/page.tsx). The settings page cannot become a Server Component itself
// without ripping out its client-side toggle logic, which is out of scope (structural fix only,
// per the mission brief). The fix is therefore a NEW server wrapper, not a change to the
// existing client component's directive:
//
//   app/admin/settings/page.tsx        <- stays 'use client', UNCHANGED apart from losing its
//                                          own top-level <main> (see below) — this is now the
//                                          form/toggle body only, rendered as a child.
//   app/admin/settings/page.tsx's default export is renamed/kept as the client toggle component,
//   OR (equally acceptable — @dev's choice) the existing default export is kept as-is and a new
//   Server Component wrapper file is introduced. Either way, exactly ONE of the following two
//   layouts must exist on disk; A1's structural check accepts either:
//
// LAYOUT OPTION A — page.tsx becomes a thin async Server Component that renders chrome plus a
// separately-named client component:
//
//   app/admin/settings/page.tsx (Server Component, async, no 'use client'):
//
import { redirect } from 'next/navigation';
import { getAdminSession, hasCapability } from '@/lib/admin-auth';
import { resolveShowWindowLookup } from '@/lib/show-window-lookup';
import { NATIONAL_SHOW_ID } from '@/lib/tickets-constants';
import { UtilityBar, Header, Footer } from '@/components/chrome';
import { sanityFetch } from '@/sanity/lib/fetch';
import { nationalShowQuery } from '@/sanity/queries';
import { AdminNav } from '@/components/admin/AdminNav';
import { OzowSandboxTestModeToggle } from '@/components/admin/OzowSandboxTestModeToggle'; // the existing toggle body, moved verbatim
import type { ShowIdentity } from '@/types';

export const dynamic = 'force-dynamic'; // same reasoning as app/admin/vendors/page.tsx: this
// page never reads cookies/headers itself once the toggle body is extracted, so Next has no
// signal to treat it as dynamic and would otherwise prerender it at build time.

export default async function AdminSettingsPage() {
  const session = await getAdminSession();
  if (!session.ok) {
    redirect('/admin/login');
  }

  const now = new Date();
  const lookupShowWindow = await resolveShowWindowLookup(NATIONAL_SHOW_ID, now);
  const canReviewVendors = hasCapability(
    session.decodedToken,
    NATIONAL_SHOW_ID,
    'review-vendor-applications',
    { now, lookupShowWindow },
  );
  const canManagePaymentSettings = hasCapability(
    session.decodedToken,
    NATIONAL_SHOW_ID,
    'manage-payment-settings',
    { now, lookupShowWindow },
  );

  const show = await sanityFetch<ShowIdentity>({
    query: nationalShowQuery,
    tags: ['nationalShow', 'sanity'],
  });

  return (
    <>
      <UtilityBar show={show} />
      <Header />
      <AdminNav
        variant="bar"
        canReviewVendors={canReviewVendors}
        canManagePaymentSettings={canManagePaymentSettings}
      />
      <OzowSandboxTestModeToggle />
      <Footer />
    </>
  );
}
